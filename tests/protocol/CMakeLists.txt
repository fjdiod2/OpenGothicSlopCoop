cmake_minimum_required(VERSION 3.16)
project(OpenGothicProtocolTests LANGUAGES CXX)
include(../../cmake/CoopIdentity.cmake)

# Standalone: no renderer, engine dependencies, submodules or Gothic assets.
option(COOP_SANITIZERS "Run protocol tests with address/undefined sanitizers" OFF)
add_executable(coop-protocol-tests protocol_test.cpp ../../common/network/protocol.cpp)
add_executable(coop-object-tests objectregistry_test.cpp ../../common/game/objectregistry.cpp)
add_executable(coop-interpolation-tests interpolation_test.cpp)
add_executable(coop-fingerprint-tests fingerprint_test.cpp ../../common/network/fingerprint.cpp)
coop_identity(coop-fingerprint-tests)
set(COOP_TEST_TARGETS coop-protocol-tests coop-object-tests coop-interpolation-tests coop-fingerprint-tests)
add_executable(coop-connection-tests connection_test.cpp ../../common/network/protocol.cpp
                 ../../common/network/session.cpp ../../common/network/fingerprint.cpp ../../common/network/tls.cpp)
list(APPEND COOP_TEST_TARGETS coop-connection-tests)
coop_identity(coop-connection-tests)
if(WIN32)
  target_link_libraries(coop-connection-tests ws2_32)
endif()
foreach(target ${COOP_TEST_TARGETS})
target_include_directories(${target} PRIVATE ../../common)
target_compile_features(${target} PRIVATE cxx_std_20)
set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX)
else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wconversion -Werror)
endif()
if(COOP_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "COOP_SANITIZERS requires Clang or GCC")
  endif()
  target_compile_options(${target} PRIVATE -fsanitize=address,undefined
                         -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined)
endif()
endforeach()

enable_testing()
add_test(NAME fingerprints COMMAND coop-fingerprint-tests)
foreach(case motion discontinuities)
  add_test(NAME interpolation-${case} COMMAND coop-interpolation-tests ${case})
  set_tests_properties(interpolation-${case} PROPERTIES TIMEOUT 20)
endforeach()
foreach(case lifecycle checkpoint malformed churn)
  add_test(NAME objects-${case} COMMAND coop-object-tests ${case})
  set_tests_properties(objects-${case} PROPERTIES TIMEOUT 20)
endforeach()
foreach(case emotes dialogue-intents story-projection golden pickup-transfers weapon-equipment consumables combat-state message-types fragments combined malformed-header malformed-payload
             encode-validation eof receive-limits write-limits partial-writes stream-stress malformed-stress)
  add_test(NAME protocol-${case} COMMAND coop-protocol-tests ${case})
  set_tests_properties(protocol-${case} PROPERTIES TIMEOUT 20)
endforeach()
foreach(case heartbeat authentication endpoints addresses connection rejections timeout binding binding-authority snapshot-roundtrip snapshot-failures input-roundtrip input-validation input-disconnect poses-roundtrip poses-validation rejoin-queues rejoin-stale save-input-hold load-generation)
  add_test(NAME connection-${case} COMMAND coop-connection-tests ${case})
  set_tests_properties(connection-${case} PROPERTIES TIMEOUT 20)
endforeach()
